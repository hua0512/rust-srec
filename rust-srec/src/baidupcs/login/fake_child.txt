use std::{env, fs, io::{self, Read}, path::Path, thread, time::Duration};
const TRACE: &str = __TRACE__;
fn main() {
    let args: Vec<_> = env::args().collect();
    if args.get(1).map(String::as_str) == Some("--descendant") {
        fs::write(format!("{TRACE}.childpid"), std::process::id().to_string()).unwrap();
        thread::sleep(Duration::from_secs(120));
        return;
    }
    let directory = env::var("BAIDUPCS_GO_CONFIG_DIR").unwrap();
    if args.get(1).map(String::as_str) == Some("env") {
        println!("BAIDUPCS_GO_CONFIG_DIR=\"{directory}\"");
        return;
    }
    fs::write(format!("{TRACE}.argv"), format!("{args:?}")).unwrap();
    fs::write(format!("{TRACE}.stage"), &directory).unwrap();
    fs::write(format!("{TRACE}.pid"), std::process::id().to_string()).unwrap();
    assert!(Path::new(&directory).join("pcs_command_history.txt").is_dir());
    let mode = Path::new(&args[0]).file_stem().unwrap().to_string_lossy();
    if mode == "descendant" || mode == "hangtree" {
        let _child = std::process::Command::new(env::current_exe().unwrap()).arg("--descendant").spawn().unwrap();
        while !Path::new(&format!("{TRACE}.childpid")).exists() { thread::sleep(Duration::from_millis(5)); }
    }
    if mode == "unread" {
        thread::sleep(Duration::from_secs(120));
        return;
    }
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    fs::write(format!("{TRACE}.stdin"), &input).unwrap();
    if mode == "hang" || mode == "hangtree" {
        thread::sleep(Duration::from_secs(120));
        return;
    }
    fs::write(Path::new(&directory).join("pcs_config.json"), r#"{"baidu_active_uid":42,"baidu_user_list":[{"uid":42,"name":"new-user","bduss":"FAKE-SAVED"}],"proxy":"do-not-import"}"#).unwrap();
    println!("echo: {input}");
    if mode != "reject" && mode != "spoof" { println!("百度帐号登录成功: new-user"); }
    if mode == "badexit" { std::process::exit(2); }
}
